// Takes the word and displays it as _
void underscoredisplay(const char word[]) {
    int size = strlen(word);

    for (int i = 0; i < size; i++) {
        printf("_ ");
    }
}

void checkForWord(const char word[], const char userinput[], int verify) {

    int size = strlen(word);

    for (int i = 0; i < size; i++) {
        if (i < strlen(userinput) && userinput[i] == word[i]) {
            printf("%c", userinput[i]);
            verify++;
            
        } else if (i >= strlen(userinput) || userinput[i] != word[i]) {
            printf("_");
        }
    }
}

void checkForCompletion(const char word[], int verify, const char userinput[]) {
    
    while(strlen(word) != verify){
        checkForWord(word, userinput, verify);
    } 

}
